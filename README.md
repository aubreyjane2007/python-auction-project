bids = {}

while True:
  name = input("Enter bidder name: ")
  bid = float(input("Enter bid amount: $"))

  bids[name] = bid

  more_bidders = input("Are there any other bidders? (yes/no): ").lower()

  if more_bidders == "no":
    break

highest_bid = 0
winner = ""

for bidder in bids:
  if bids[bidder] > highest_bid:
    highest_bid = bids[bidder]
    winner = bidder

print("\nAuction Results")
print("Winner:", winner)
print("Winning Bid: $", highest_bid)
